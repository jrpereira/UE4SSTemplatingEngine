cmake_minimum_required(VERSION 3.22)
project(ModCoreTemplatesStartup VERSION 0.0.20 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
    add_executable(LoopStartQueueTests ../tests/native/LoopStartQueueTests.cpp)
    target_include_directories(LoopStartQueueTests PRIVATE include)
    add_test(NAME LoopStartQueueTests COMMAND LoopStartQueueTests)
    find_package(Threads REQUIRED)
    add_executable(LifetimeIndexTests ../tests/native/LifetimeIndexTests.cpp)
    target_include_directories(LifetimeIndexTests PRIVATE include)
    target_link_libraries(LifetimeIndexTests PRIVATE Threads::Threads)
    add_test(NAME LifetimeIndexTests COMMAND LifetimeIndexTests)
    add_executable(TemplateRegistryTests ../tests/native/TemplateRegistryTests.cpp)
    target_include_directories(TemplateRegistryTests PRIVATE include)
    target_link_libraries(TemplateRegistryTests PRIVATE Threads::Threads)
    add_test(NAME TemplateRegistryTests COMMAND TemplateRegistryTests)
endif()
if(WIN32)
    if(NOT MSVC)
        message(FATAL_ERROR "The UE4SS 97b7e501 ABI requires MSVC.")
    endif()
    find_program(MCT_LIB_EXE lib.exe REQUIRED)
    set(MCT_IMPORT_LIB "${CMAKE_CURRENT_BINARY_DIR}/UE4SS-97b7e501.lib")
    add_custom_command(OUTPUT "${MCT_IMPORT_LIB}"
        COMMAND "${MCT_LIB_EXE}" /nologo /machine:x64
            "/def:${CMAKE_CURRENT_SOURCE_DIR}/abi/UE4SS-97b7e501.def" "/out:${MCT_IMPORT_LIB}"
        DEPENDS abi/UE4SS-97b7e501.def VERBATIM)
    add_custom_target(MCTImportLibrary DEPENDS "${MCT_IMPORT_LIB}")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Scripts/mct/loop_start.lua" MCT_LOOP_START_LUA)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ../Scripts/mct/loop_start.lua)
    configure_file(LoopStartLua.hpp.in generated/LoopStartLua.hpp @ONLY)
    add_library(ModCoreTemplatesStartup SHARED src/LoopStartMod.cpp)
    add_dependencies(ModCoreTemplatesStartup MCTImportLibrary)
    target_include_directories(ModCoreTemplatesStartup PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(ModCoreTemplatesStartup PRIVATE "${MCT_IMPORT_LIB}" kernel32)
    target_compile_options(ModCoreTemplatesStartup PRIVATE /W4 /WX /GR- /permissive-)
    set_property(TARGET ModCoreTemplatesStartup PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
    set_target_properties(ModCoreTemplatesStartup PROPERTIES OUTPUT_NAME main PREFIX ""
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/dist/_ModCore_Templates/dlls")
endif()
